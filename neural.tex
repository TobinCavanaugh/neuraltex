\def\backpropogate#1{
    
    \i=0


    \global\tmpw=\numexpr \getw{\the\i} + \eta * \err * \gettruth{#1}{\the\i} \relax
    \setw{\the\i}{\the\tmpw}
    
    \advance\i by 1
    
    \global\tmpw=\numexpr \getw{\the\i} + \eta * \err * \gettruth{#1}{\the\i} \relax
    \setw{\the\i}{\the\tmpw}
    

    \global\tmpb=\numexpr \b + \eta * \err \relax
    \global\b=\the\tmpb

}

\newcount\target


\def\procrow#1{%
    \act=\numexpr (\gettruth{#1}{0} * \getw{0}) + (\gettruth{#1}{1} * \getw{1}) + \the\b \relax 

    \ifnum\act<0
        \prediction=0
    \else
        \prediction=1
    \fi

    \target=\gettruth{#1}{2}
    \err=\numexpr \target - \prediction \relax

    \ifnum\err=0
    \else 
        \converged=0
        \backpropogate{#1}
    \fi

}

\def\converge{%
    \procrow{0}%
    \procrow{1}%
    \procrow{2}%
    \procrow{3}%
}
